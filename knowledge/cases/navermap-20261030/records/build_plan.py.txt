    if 'sea' in ident:
        m=(p[:,:,2]-p[:,:,0]>65)&(p[:,:,0]<80)&(p[:,:,1]<170)
        radius=3
    else:
        m=(p.max(2)<155)
        radius=3
    local=Image.fromarray((m*255).astype('uint8')).filter(ImageFilter.MaxFilter(radius*2+1))

# Solve a local harmonic extension, only inside the recorded edit mask.
unknown=mask>0
b=a.copy(); known=~unknown
# Propagate trustworthy boundary colors inward.
filled=known.copy()
for i in range(110):
    count=np.zeros(filled.shape,float); total=np.zeros_like(b)
    for dy,dx in ((-1,0),(1,0),(0,-1),(0,1)):
        f=np.roll(filled,(dy,dx),(0,1)); count+=f
        total+=np.roll(b,(dy,dx),(0,1))*f[:,:,None]
    frontier=(~filled)&(count>0)
    if not frontier.any():break
    b[frontier]=total[frontier]/count[frontier,None];filled[frontier]=True
for i in range(450):
    mean=(np.roll(b,1,0)+np.roll(b,-1,0)+np.roll(b,1,1)+np.roll(b,-1,1))/4
    b[unknown]=mean[unknown]
Image.fromarray(np.clip(b,0,255).astype('uint8')).save(ROOT/'clean-background.png')
