from pathlib import Path
from PIL import Image,ImageDraw,ImageFont
import json
root=Path(__file__).resolve().parent
job=json.loads((root/'job.json').read_text('utf-8'))
t=job['targets'][0]; assets=[]; mapping=[]; ids={}
(root/'lettering-assets').mkdir(exist_ok=True)
for row in t['rows']:
    text=row['text'];s=3
    f=ImageFont.truetype(row['font_path'],round(row['font_size']*s))
    stroke=round(row['stroke']*s)
    box=f.getbbox(text,stroke_width=stroke)
    tile=Image.new('RGBA',(box[2]-box[0]+12,box[3]-box[1]+12))
    ImageDraw.Draw(tile).text((6-box[0],6-box[1]),text,font=f,fill=row['color'],stroke_width=stroke,stroke_fill='white')
    tile=tile.crop(tile.getbbox())
    tile=tile.resize((round(tile.width/s),round(tile.height/s)),Image.Resampling.LANCZOS)
    name=f"lettering-assets/{row['id']}.png";tile.save(root/name)
    position=[round(row['center'][0]-tile.width/2),round(row['center'][1]-tile.height/2)]
    assets.append({'id':row['id'],'path':name,'kind':'text','position':position})
    ids[row['id']]=text
for m in t['region_map']:
    mapping.append({'region_id':m['region_id'],'row_ids':[],'asset_texts':[{'asset_id':i,'text':ids[i]} for i in m['row_ids']]})
t.update(rows=[],assets=assets,region_map=mapping)
(root/'job-assets-v3.json').write_text(json.dumps(job,ensure_ascii=False,indent=2),'utf-8')
print('Generated white-outline Chinese RGBA text assets.')
